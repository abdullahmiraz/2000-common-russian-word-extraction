% Entry Macro for Dense, Readable 2-Column A4 Layout
\newcommand{\vocentry}[6]{%
  \noindent
  \begin{minipage}{\linewidth}
    \raggedright
    {\footnotesize\sffamily\bfseries\color{white}\colorbox{badgebg}{\rule[-0.15ex]{0pt}{1.8ex}\hspace{0.2em}#1\hspace{0.2em}}}%
    \hspace{0.35em}%
    {\normalsize\bfseries\color{primary}#2}%
    \hspace{0.3em}%
    {\ipafont\footnotesize\color{secondary}[#3]}%
    \hspace{0.25em}%
    \textcolor{bordercol}{\textbf{--}}%
    \hspace{0.25em}%
    {\footnotesize\bfseries\itshape\color{darktext}#4}\par
    \vspace{0.18em}%
    \hangindent=0.8em\hangafter=1\footnotesize
    \noindent\textcolor{darktext}{#5}\par
    \vspace{0.08em}%
    \noindent\textcolor{mutedtext}{\textit{#6}}%
  \end{minipage}%
  \vspace{0.5em}\par
}
