% Thick Border Box for Serial Number
\newcommand{\entrybadge}[1]{%
  \begingroup
  \setlength{\fboxrule}{1.2pt}%
  \setlength{\fboxsep}{2.2pt}%
  \fcolorbox{boxborder}{white}{\footnotesize\sffamily\bfseries\color{secondary}\strut #1}%
  \endgroup
}

% Optimized Entry Macro: Larger readable fonts, thick bordered numbers, compact flow
\newcommand{\vocentry}[6]{%
  \noindent
  \begin{minipage}{\linewidth}
    \raggedright
    \entrybadge{#1}%
    \hspace{0.35em}%
    {\large\bfseries\color{primary}#2}%
    \hspace{0.3em}%
    {\ipafont\small\color{secondary}[#3]}%
    \hspace{0.25em}%
    \textcolor{bordercol}{\textbf{--}}%
    \hspace{0.25em}%
    {\small\bfseries\itshape\color{darktext}#4}\par
    \vspace{0.18em}%
    \hangindent=0.8em\hangafter=1\small
    \noindent\textcolor{darktext}{#5}\par
    \vspace{0.08em}%
    \noindent\textcolor{mutedtext}{\textit{#6}}%
  \end{minipage}%
  \vspace{0.55em}\par
}
